\subsection*{Answers}

\subsubsection*{Question 1}

Since $\angle FHG$ is complementary to 64\degree,

$\angle FHG = 26\degree$.

Find $\angle FGH$ using sine rule,

\begin{equation*}
\begin{split}
\frac{\sin(\angle FGH)}{83} &= \frac{\sin(26\degree)}{47} \\
\sin(\angle FGH) &= \frac{\sin(26\degree)}{47}\times 83 \\
\angle FGH &\approx 50.7\degree
\end{split}
\end{equation*}

But since $\angle FGH$ is obtuse,

$\angle FGH \approx 180\degree - 50.7\degree = 129.3\degree$

$\angle GFH = 180\degree - 129.3\degree - 26\degree = 24.7\degree$

\begin{equation*}
\begin{split}
\frac{GH}{\sin(24.7\degree)} &= \frac{47}{\sin(26\degree)} \\
GH &= \frac{47}{\sin(26\degree)}\times \sin(24.7\degree)\\
&\approx 44.80
\end{split}
\end{equation*}

\subsubsection*{Question 2}

The area of $\triangle PMN$ is 

\begin{equation*}
A_{\triangle PMN} = \frac{1}{2}\times 4\times 8\times \sin(110\degree)
\approx 15.04\mathrm{m}^2
\end{equation*}

The length of $PN$
\begin{equation*}
\begin{split}
PN^2 &= 4^2 + 8^2 - 2\times 4\times 8\times\cos(110\degree) \\
PN &= \sqrt{101.889...} \\
&\approx 10.1\mathrm{m}
\end{split}
\end{equation*}

The area of equilateral $\triangle PNQ$
\begin{equation*}
A_{\triangle PNQ} = \frac{1}{2}\times 10.1^2 \times \sin(60\degree)
\approx 44.17\mathrm{m}^2
\end{equation*}

The cost is
$$(15.04 + 44.17)\times 45 = \$2664.45$$

\subsubsection*{Question 3}

\begin{figure}[h]
\centering
\begin{tikzpicture}[scale=0.3]
\tikzset{
}
\draw (0,0) +(-2,0) -- +(2,0) +(0,-2) -- +(0,2);
\draw (75:15) +(-2,0) -- +(2,0) +(0,-2) -- +(0,2);
\draw (0,0) +(90:0.3) arc[radius=0.3,start angle=90,delta angle=-260] 
node[pos=0.5,below right] {\footnotesize260\degree};
\draw (90:0.6) arc[radius=0.6,start angle=90,delta angle=-15];
\draw (82.5:0.6) -- (-1,2) node[left]{\footnotesize015\degree};
\draw (75:15) +(90:0.3) arc[radius=0.3,start angle=90,delta angle=-290]
node[pos=0.6,below right]{\footnotesize290\degree};
\draw (-170:10) node[left] {$B$} -- 
node[pos=0.5,below=0.2cm] {10km} 
(0:0) node[above right] {$A$} -- 
(75:15) node[above right] {$A'$} -- node[pos=0.5,above=0.2cm]{16km}
+(170:16) node[left]{$B'$};
\draw[dashed] (-170:10) -- (75:15) +(170:16) -- (-170:10);
\end{tikzpicture}
\end{figure}

The length of $AA'$ is $5\times 3=15$.

Find the angle $\angle BAA'$, $360\degree - 260\degree + 15\degree
= 115\degree$.

Use the cosine rule to find the length of $A'B$,

\begin{equation*}
\begin{split}
A'B &= \sqrt{10^2+15^2-2\times 10\times 15\times \cos(115\degree)} \\
&\approx 21.26
\end{split}
\end{equation*}

Find $\angle BA'A$ using cosine rule

\begin{equation*}
\begin{split}
\angle BA'A &= \cos^{-1}\left(\frac{15^2+21.26^2-10^2}
{2\times 15 \times 21.26}\right) \\
&\approx 25.22\degree
\end{split}
\end{equation*}

Find the angle of $B'A'B$ is $290\degree - 180\degree - 15\degree
- 25.22\degree = 69.78\degree$

Use the cosine rule to find $BB'$,
\begin{equation*}
\begin{split}
B'B &= \sqrt{16^2+21.26^2-2\times 16\times 21.26\times \cos(69.78\degree)} \\
&\approx 21.75
\end{split}
\end{equation*}

The speed of Ship 2 is $21.75\div 3 = 7.25\mathrm{km/hr}$.
